-- [1. حماية الميتا-تيبل ومنع الـ Logs والـ Kick]
pcall(function()
    local gmt = getrawmetatable(game)
    setreadonly(gmt, false)
    local old = gmt.__namecall
    gmt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        if method == "FireServer" and (tostring(self):find("Log") or tostring(self):find("Admin") or tostring(self):find("Report")) then 
            return nil 
        end
        if method == "Kick" then return nil end
        return old(self, ...)
    end)
    setreadonly(gmt, true)
end)

-- [4. قسم الـ FE BANG]
local bangInput = Instance.new("TextBox", MainFrame)
bangInput.Size = UDim2.new(0.9, 0, 0, 35)
bangInput.PlaceholderText = "Write Player Name..."
bangInput.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
bangInput.TextColor3 = Color3.new(1, 1, 1)

local bangLoop
createButton("START BANGING", Color3.fromRGB(200, 0, 100), function()
    local target = nil
    for _, v in pairs(game.Players:GetPlayers()) do
        if v ~= player and v.Name:lower():find(bangInput.Text:lower()) then target = v break end
    end
    if target and target.Character then
        if bangLoop then bangLoop:Disconnect() end
        local anim = Instance.new("Animation")
        anim.AnimationId = "rbxassetid://148840371"
        local load = player.Character.Humanoid:LoadAnimation(anim)
        load:Play()
        load:AdjustSpeed(5)
        bangLoop = game:GetService("RunService").Heartbeat:Connect(function()
            pcall(function() player.Character.HumanoidRootPart.CFrame = target.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, 1.1) end)
        end)
    end
end)

createButton("STOP BANG", Color3.fromRGB(80, 80, 80), function()
    if bangLoop then bangLoop:Disconnect() end
    for _, v in pairs(player.Character.Humanoid:GetPlayingAnimationTracks()) do v:Stop() end
end)
